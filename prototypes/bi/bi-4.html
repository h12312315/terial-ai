Total Revenue by Region
SQL editor
SELECT
  r.region_name        AS "Region",
  SUM(o.value)         AS "Total Revenue",
  COUNT(o.id)          AS "Deal Count",
  AVG(o.value)         AS "Avg Deal Size",
  ROUND(
    SUM(CASE WHEN o.status = 'won' THEN 1 ELSE 0 END)::numeric
    / NULLIF(COUNT(o.id), 0) * 100, 0
  )                    AS "Win Rate"
FROM opportunities o
JOIN regions r ON r.id = o.region_id
WHERE o.closed_at >= DATE_TRUNC('quarter', CURRENT_DATE)
GROUP BY r.region_name
ORDER BY "Total Revenue" DESC;
